document.addEventListener("DOMContentLoaded", function () {
function loadTopBanner() {
let ad = document.querySelector(".adsbygoogle.top-banner");
if (ad && !ad.dataset.adLoaded) {
(adsbygoogle = window.adsbygoogle || []).push({});
ad.dataset.adLoaded = "true";
}
}
function lazyLoadAds() {
let adSlots = document.querySelectorAll(".adsbygoogle:not(.top-banner):not(.no-lazy)");
if (window.innerWidth > 741) {
adSlots.forEach(ad => {
if (!ad.dataset.adLoaded) {
(adsbygoogle = window.adsbygoogle || []).push({});
ad.dataset.adLoaded = "true";
}
});
return;
}
let observer = new IntersectionObserver(
(entries, observer) => {
entries.forEach(entry => {
if (entry.isIntersecting) {
let ad = entry.target;
if (!ad.dataset.adLoaded) {
(adsbygoogle = window.adsbygoogle || []).push({});
ad.dataset.adLoaded = "true";
}
observer.unobserve(ad);
}
});
},
{ rootMargin: "50px 0px", threshold: 0.1 }
);
adSlots.forEach(ad => observer.observe(ad));
}
function loadAdsenseScript(callback) {
if (window.adsbygoogle) {
callback();
return;
}
let script = document.createElement("script");
script.src = "https://pagead2.googlesyndication.com/pagead/js/adsbygoogle.js";
script.async = true;
script.setAttribute("data-ad-client", "ca-pub-6100133717761396");
script.crossOrigin = "anonymous";
script.onload = callback;
document.head.appendChild(script);
}
loadAdsenseScript(function() {
loadTopBanner();
lazyLoadAds();
});
});
