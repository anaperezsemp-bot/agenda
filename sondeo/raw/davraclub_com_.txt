    function showToast(message) {
        const toast = document.createElement('div');
        toast.className = 'toast';
        toast.innerHTML = message;
        document.body.appendChild(toast);
        
        setTimeout(() => {
            toast.classList.add('show');
        }, 100);
        
        setTimeout(() => {
            toast.classList.remove('show');
            setTimeout(() => {
                document.body.removeChild(toast);
            }, 300);
        }, 3000);
    }
    const newsletterBar = document.querySelector('.newsletter-bar');
    const newsletterForm = document.querySelector('.newsletter-form');
    const closeForm = document.querySelector('.newsletter-form-close');    
        if (newsletterBar) {
            newsletterBar.addEventListener('click', function() {
                newsletterForm.style.display = 'flex';
            });
        }
        if (closeForm) {
            closeForm.addEventListener('click', function() {
                newsletterForm.style.display = 'none';
            });
        }
        // Cerrar formulario al hacer clic fuera
        newsletterForm.addEventListener('click', function(e) {
            if (e.target === this) {
                this.style.display = 'none';
            }
        });
        // Validación del formulario
        const subscribeForm = document.getElementById('subscribeForm'); 
        if (subscribeForm) {
            subscribeForm.addEventListener('submit', function(e) {
                e.preventDefault();
                
                const emailInput = this.querySelector('input[type="email"]');
                const email = emailInput.value.trim();
                
                if (!email) {
                    showToast('Por favor, introduce tu email');
                    return;
                }
                
                // Enviar datos al servidor
                fetch('/subscribe', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/x-www-form-urlencoded',
                    },
                    body: `email=${encodeURIComponent(email)}`
                })
                .then(response => response.json())
                .then(data => {
                    if (data.success) {
                        showToast(data.message);
                        emailInput.value = '';
                        newsletterForm.style.display = 'none';
                    } else {
                        showToast(data.message || 'Ha ocurrido un error');
                    }
                })
                .catch(error => {
                    console.error('Error:', error);
                    showToast('Ha ocurrido un error');
                });
            });
        }                                     
    </script>
